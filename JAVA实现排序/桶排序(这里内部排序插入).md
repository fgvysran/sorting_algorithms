class Solution {
    public int[] sortArray(int[] nums) {
        bucketSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void bucketSort(int[]a,int l,int r){
        //折中根号n个桶，理论n 个最好
        int n = r - l + 1,mx = Integer.MIN_VALUE,mn = Integer.MAX_VALUE;
        if(n <= 1)return;
        int cnt = (int)Math.sqrt(n);
        for(int i = l;i <= r;i++){
            mx = Math.max(mx,a[i]);
            mn = Math.min(mn,a[i]);
        }
        if(mx == mn)return;
        int offset = -mn;
        List<Integer>[] bucket = new ArrayList[cnt];
        for(int i = 0;i < cnt;i++)bucket[i] = new ArrayList<>();
        //装桶
        for(int i = l;i <= r;i++){
            int ind = ((long)(a[i] + offset) * (cnt - 1) / (mx -mn));//归一化装桶
            bucket[ind].add(a[i]);
        }
        //桶内排序(这里插入排序，也可以快速排序啥的)
        for(int i = 0;i < cnt;i++){
            if(!bucket[i].isEmpty()){
                insertSort(bucket[i]);
            }
        }
        int p = l;
        for(int i = 0;i < cnt;i++){
            for(int x:bucket[i]){
                a[p++] = x;
            }
        }
    }
    void insertSort(List<Integer> arr){
        int n = arr.size();
        for(int i = 1;i < n;i++){
            int j = i - 1,t = arr.get(i);
            while(j >= 0 && arr.get(j) > t){
                arr.set(j + 1,arr.get(j));j--;
            }
            arr.set(j + 1,t);
        }
    }
}
