class Solution {
    public int[] sortArray(int[] nums) {
        radixSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void radixSort(int[]a,int l,int r){
        //假设不会溢出加上偏移量
        int mx = Integer.MIN_VALUE,mn = Integer.MAX_VALUE;
        //o(n)找最大值，最小值(因为有负数)
        for(int i = l;i <= r;i++){
            mx = Math.max(a[i],mx);
            mn = Math.min(mn,a[i]);    
        }
        int offset = mn >= 0?0:-mn;
        for(int exp = 1;(mx + offset) / exp > 0;exp *= 10){
            continueSortBase10(a,l,r,exp,offset);
        }

    }
    void continueSortBase10(int[]a,int l,int r,int exp,int offset){
        int[]cnt = new int[10];
        int[]help = new int[r - l + 1];//辅助数组
        for(int i = l;i <= r;i++){
            int d = (a[i] + offset) / exp % 10;
            cnt[d]++;
        }
        for(int i = 1;i < 10;i++)cnt[i] += cnt[i - 1];//前缀和
        for(int i = r;i >= l;i--){
            int d = (a[i] + offset) / exp % 10;
            help[--cnt[d]] = a[i];
        }
        for(int i = 0;i < help.length;i++)a[l + i] = help[i];
        return;
    }
}
