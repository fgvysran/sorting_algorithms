class Solution {
    public int[] sortArray(int[] nums) {
        insertSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void insertSort(int[]a,int l,int r){
        for(int i = l + 1;i <= r;i++){
            int j = i - 1,t = a[i];
            while(j >= l && a[j] > t){
                a[j + 1] = a[j];
                j--;
            }
            a[j + 1] = t;
        }
    }
}
