class Solution {
    public int[] sortArray(int[] nums) {
        heapSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void heapSort(int[]a,int l,int r){
        int offset = -l;
        int n = r - l + 1;
        //建堆
        for(int k = n / 2 - 1;k >= 0;k--){
            int ind = k - offset;
            down(a,ind,r,offset);
        }
        //每次取堆顶
        for(int k = n - 1;k > 0;k--){
            int t = a[l];a[l] = a[k - offset];a[k - offset] = t;
            down(a,l,k - 1 - offset,offset);
        }
        return ;
    }
    void down(int[]a,int p,int last,int offset){
        p += offset;last += offset;//对应[0,..]
        while(p <= last){
            int ind = p - offset;
            int son1 = p * 2 + 1,son2 = p * 2 + 2; 
            if(son1 <= last && a[son1 - offset] > a[ind])ind = son1 - offset;
            if(son2 <= last && a[son2 - offset] > a[ind])ind = son2 - offset;
            if(ind != p - offset){
                int t = a[p - offset];a[p - offset] = a[ind];a[ind] = t;
                p = ind;
            }else{
                break;
            }
        }
    }
    
}
