class Solution {
    public int[] sortArray(int[] nums) {
        quickSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void quickSort(int[]a,int l,int r){
        if(r - l <= 0)return ;//数据大小<= 1直接返回
        //中间点//此处简单实现就取l，还有多种取中间点方法
        //然后把中间点和最左边点交换，最左边点空缺
        int num = a[l];
        int p1 = l,p2 = r;
        while(p1 < p2){
            //填左边空缺，找一个小于num
            while(p1 < p2 && a[p2] >= num)p2--;
            a[p1] = a[p2];//右边空缺
            //填y右边空缺，找一个大于num
            while(p1 < p2 && a[p1] <= num)p1++;
            a[p2] = a[p1];//左边空缺
        }
        //填补最后一个空缺
        a[p1] = num;
        //递归排序子范围[l,p1 - 1],[p1 + 1,r] 
        quickSort(a,l,p1 - 1);
        quickSort(a,p1 + 1,r);
    }
}
