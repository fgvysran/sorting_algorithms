class Solution {
    public int[] sortArray(int[] nums) {
        selectionSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void selectionSort(int[]a,int l,int r){
        for(int i = r;i > l;i--){
            int mx = l;
            for(int j = l + 1;j <= i;j++){
                if(a[mx] < a[j])mx = j;
            }
            int t = a[mx];a[mx] = a[i];a[i] = t;
        }
    }

}
