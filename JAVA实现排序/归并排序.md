class Solution {
    public int[] sortArray(int[] nums) {
        mergeSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void mergeSort(int[]a,int l,int r){
        if(r - l <= 0)return ;//<=1个元素直接返回
        //辅助数组
        int[]help = new int[r - l + 1];
        int mid = (l + r) >> 1;
        //先分
        mergeSort(a,l,mid);
        mergeSort(a,mid + 1,r);
        //再和
        int p1 = l,p2 = mid + 1,p = 0;
        while(p1 <= mid || p2 <= r){
            if(p2 > r || p1 <= mid && a[p1] <= a[p2]){
                help[p++] = a[p1++];
            }else{
                help[p++] = a[p2++];
            }
        }
        for(int i = 0;i < p;i++){
            a[l + i] = help[i];
        }
        return ;
    }
}
