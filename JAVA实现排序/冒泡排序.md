class Solution {
    public int[] sortArray(int[] nums) {
        bubleSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    void bubleSort(int[]a,int l,int r){
        for(int i = r;i > l;i--){
            boolean f = true;
            for(int j = l;j < i;j++){
                if(a[j + 1] < a[j]){
                    f = false;
                    int t = a[j + 1];a[j + 1] = a[j];a[j] = t;
                }
            }
            if(f)return;
        }
    }

}
