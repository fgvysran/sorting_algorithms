class Solution {
    public int[] sortArray(int[] nums) {
        shellSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    //折半增量gap
    void shellSort(int[]a,int l,int r){
        int n = r - l + 1;
        if(n <= 1)return ;
        for(int gap = n / 2;gap > 0;gap /= 2){
            for(int i = l + gap;i <= r;i ++){
                int j = i  - gap,t = a[i];
                while(j >= l && a[j] > t){
                    a[j + gap] = a[j];j -= gap;
                }
                a[j + gap] = t;
            }
        }
    }
}

class Solution {
    public int[] sortArray(int[] nums) {
        shellSort(nums,0,nums.length - 1);
        return nums;
    }
    //[l,r]
    //Hibbard gap增量
    //(2^k-1)，相邻增量互质,最坏O(n^1.5)，比折半更好一点
    void shellSort(int[]a,int l,int r){
        int gap = 1,n = r - l + 1;
        while(gap <= n / 2)gap = gap * 2 + 1;
        for(;gap > 0;gap /= 2){
            for(int i = l + gap;i <= r;i++){
                int j = i - gap,t = a[i];
                while(j >= l && a[j] > t){
                    a[j + gap] = a[j];j -= gap;
                }
                a[j + gap] = t;
            }
        }
    }
}
